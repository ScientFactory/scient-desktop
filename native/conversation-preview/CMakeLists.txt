cmake_minimum_required(VERSION 3.21)
project(scient_conversation_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC AND VCPKG_TARGET_TRIPLET MATCHES "^(x64|arm64)-windows-static$")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(APPLE)
  if(NOT SCIC_JSON_C_ROOT OR NOT EXISTS "${SCIC_JSON_C_ROOT}/lib/libjson-c.a" OR
     NOT EXISTS "${SCIC_JSON_C_ROOT}/include/json-c/json.h")
    message(FATAL_ERROR "Set SCIC_JSON_C_ROOT to the isolated pinned json-c static install")
  endif()
  if(NOT LibArchive_LIBRARY OR NOT LibArchive_INCLUDE_DIR)
    message(FATAL_ERROR "Set LibArchive_LIBRARY to the SDK stub and LibArchive_INCLUDE_DIR to pinned headers")
  endif()
  add_library(scic_json_c STATIC IMPORTED)
  set_target_properties(scic_json_c PROPERTIES
    IMPORTED_LOCATION "${SCIC_JSON_C_ROOT}/lib/libjson-c.a"
    INTERFACE_INCLUDE_DIRECTORIES "${SCIC_JSON_C_ROOT}/include")
  set(SCIC_JSON_TARGET scic_json_c)
endif()
find_package(LibArchive REQUIRED)
if(NOT APPLE AND NOT WIN32)
  find_package(OpenSSL REQUIRED)
endif()
if(NOT APPLE)
  find_package(json-c CONFIG QUIET)
  if(TARGET json-c::json-c)
    set(SCIC_JSON_TARGET json-c::json-c)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JSON_C REQUIRED IMPORTED_TARGET json-c)
    set(SCIC_JSON_TARGET PkgConfig::JSON_C)
  endif()
endif()

add_library(scic_preview STATIC src/scic_preview.cpp)
target_include_directories(scic_preview PUBLIC include)
target_link_libraries(scic_preview PUBLIC LibArchive::LibArchive ${SCIC_JSON_TARGET})
if(WIN32)
  target_link_libraries(scic_preview PUBLIC bcrypt)
elseif(NOT APPLE)
  target_link_libraries(scic_preview PUBLIC OpenSSL::Crypto)
endif()
set_target_properties(scic_preview PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_executable(scic-preview src/main.cpp)
target_link_libraries(scic-preview PRIVATE scic_preview)

if(APPLE)
  add_custom_command(TARGET scic-preview POST_BUILD
    COMMAND /usr/bin/codesign --force --sign - "$<TARGET_FILE:scic-preview>"
    VERBATIM)
endif()

if(WIN32)
  set(SCIC_PREVIEW_CLSID "{E0C925A3-E41D-4969-B093-4B6B16028463}" CACHE STRING
      "Channel-specific Scient conversation preview CLSID")
  string(LENGTH "${SCIC_PREVIEW_CLSID}" SCIC_GUID_LENGTH)
  if(NOT SCIC_GUID_LENGTH EQUAL 38)
    message(FATAL_ERROR "SCIC_PREVIEW_CLSID must be a braced GUID")
  endif()
  foreach(POSITION 0 9 14 19 24 37)
    string(SUBSTRING "${SCIC_PREVIEW_CLSID}" ${POSITION} 1 SCIC_GUID_DELIMITER_${POSITION})
  endforeach()
  if(NOT SCIC_GUID_DELIMITER_0 STREQUAL "{" OR
     NOT SCIC_GUID_DELIMITER_9 STREQUAL "-" OR
     NOT SCIC_GUID_DELIMITER_14 STREQUAL "-" OR
     NOT SCIC_GUID_DELIMITER_19 STREQUAL "-" OR
     NOT SCIC_GUID_DELIMITER_24 STREQUAL "-" OR
     NOT SCIC_GUID_DELIMITER_37 STREQUAL "}")
    message(FATAL_ERROR "SCIC_PREVIEW_CLSID must be a braced GUID")
  endif()
  string(SUBSTRING "${SCIC_PREVIEW_CLSID}" 1 8 SCIC_GUID_1)
  string(SUBSTRING "${SCIC_PREVIEW_CLSID}" 10 4 SCIC_GUID_2)
  string(SUBSTRING "${SCIC_PREVIEW_CLSID}" 15 4 SCIC_GUID_3)
  string(SUBSTRING "${SCIC_PREVIEW_CLSID}" 20 4 SCIC_GUID_4)
  string(SUBSTRING "${SCIC_PREVIEW_CLSID}" 25 12 SCIC_GUID_5)
  foreach(PART SCIC_GUID_1 SCIC_GUID_2 SCIC_GUID_3 SCIC_GUID_4 SCIC_GUID_5)
    if(NOT "${${PART}}" MATCHES "^[0-9A-Fa-f]+$")
      message(FATAL_ERROR "SCIC_PREVIEW_CLSID must be a braced GUID")
    endif()
  endforeach()
  foreach(INDEX RANGE 0 1)
    math(EXPR OFFSET "${INDEX} * 2")
    string(SUBSTRING "${SCIC_GUID_4}" ${OFFSET} 2 SCIC_GUID_BYTE_${INDEX})
  endforeach()
  foreach(INDEX RANGE 0 5)
    math(EXPR OFFSET "${INDEX} * 2")
    math(EXPR TARGET_INDEX "${INDEX} + 2")
    string(SUBSTRING "${SCIC_GUID_5}" ${OFFSET} 2 SCIC_GUID_BYTE_${TARGET_INDEX})
  endforeach()
  configure_file(windows/preview_identity.h.in
    "${CMAKE_CURRENT_BINARY_DIR}/preview_identity.h" @ONLY)
  add_library(ScientConversationPreview SHARED windows/preview_handler.cpp)
  target_include_directories(ScientConversationPreview PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(ScientConversationPreview PRIVATE scic_preview ole32 uuid)
  set_target_properties(ScientConversationPreview PROPERTIES PREFIX "")

  add_executable(scic-preview-com-smoke windows/com_smoke.cpp)
  target_include_directories(scic-preview-com-smoke PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(scic-preview-com-smoke PRIVATE ole32 uuid)
  enable_testing()
  add_test(NAME scic-preview-com-smoke
    COMMAND "$<TARGET_FILE:scic-preview-com-smoke>" "$<TARGET_FILE:ScientConversationPreview>")
endif()
